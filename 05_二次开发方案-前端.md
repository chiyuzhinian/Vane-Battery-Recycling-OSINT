```tsx
// src/app/collect/page.tsx

'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { useCollectTask } from '@/lib/hooks/useCollectTask';

type CollectStep = 'select-companies' | 'select-dimensions' | 'configure' | 'running' | 'complete';

export default function CollectPage() {
  const router = useRouter();
  const [step, setStep] = useState<CollectStep>('select-companies');
  const [selectedCompanies, setSelectedCompanies] = useState<string[]>([]);
  const [selectedDimensions, setSelectedDimensions] = useState<string[]>([]);
  const [config, setConfig] = useState({
    mode: 'balanced',
    quality_threshold: 70,
    enabled_sources: ['official', 'industry', 'news', 'patent', 'bidding']
  });

  const { startCollect, task, progress } = useCollectTask();

  const handleStartCollect = async () => {
    const result = await startCollect({
      companies: selectedCompanies,
      dimensions: selectedDimensions,
      ...config
    });
    
    if (result.success) {
      setStep('running');
    }
  };

  if (step === 'running') {
    return <CollectProgressView task={task} progress={progress} />;
  }

  if (step === 'complete') {
    return <CollectCompleteView task={task} />;
  }

  return (
    <div className="collect-container">
      <h1>创建采集任务</h1>
      
      {/* 步骤指示器 */}
      <StepIndicator 
        steps={['选择企业', '选择维度', '配置参数', '进行中', '完成']}
        currentStep={step}
      />

      {/* Step 1: 选择企业 */}
      {step === 'select-companies' && (
        <SelectCompaniesStep 
          selected={selectedCompanies}
          onSelect={setSelectedCompanies}
          onNext={() => setStep('select-dimensions')}
        />
      )}

      {/* Step 2: 选择维度 */}
      {step === 'select-dimensions' && (
        <SelectDimensionsStep
          selected={selectedDimensions}
          onSelect={setSelectedDimensions}
          onNext={() => setStep('configure')}
          onBack={() => setStep('select-companies')}
        />
      )}

      {/* Step 3: 配置参数 */}
      {step === 'configure' && (
        <ConfigureStep
          config={config}
          onConfigChange={setConfig}
          onStart={handleStartCollect}
          onBack={() => setStep('select-dimensions')}
        />
      )}
    </div>
  );
}

// 进度显示组件
function CollectProgressView({ task, progress }) {
  return (
    <div className="progress-container">
      <h2>{task.companies.join('、')}数据采集进行中</h2>
      
      <div className="progress-stages">
        {/* 需求拆解 */}
        <ProgressStage
          name="需求拆解"
          status={progress.stage >= 1 ? 'complete' : 'pending'}
        />
        
        {/* 多源搜证 */}
        <ProgressStage
          name="多源搜证"
          status={progress.stage === 2 ? 'running' : progress.stage > 2 ? 'complete' : 'pending'}
          progress={progress.stage === 2 ? progress.search_progress : undefined}
          details={progress.search_details}
        />
        
        {/* 数据验证 */}
        <ProgressStage
          name="数据验证"
          status={progress.stage >= 3 ? 'running' : 'pending'}
        />
        
        {/* 报告生成 */}
        <ProgressStage
          name="报告生成"
          status={progress.stage >= 4 ? 'complete' : 'pending'}
        />
      </div>

      <div className="progress-stats">
        <p>预计剩余时间: {progress.estimated_time_remaining}分钟</p>
        <p>已处理: {progress.completed_items}/{progress.total_items}</p>
      </div>

      <button onClick={() => router.push('/dashboard')}>返回Dashboard</button>
    </div>
  );
}
```

export function DataTable({ data, columns, sortable, filterable }: DataTableProps) {
  return (
    <table className="data-table">
      <thead>

          {columns.map(col => (
            <th key={col.key}>
              {col.label}
              {sortable && <SortIcon />}
            </th>
          ))}
          <th>数据质量</th>
          <th>操作</th>
        </tr>
      </thead>
      <tbody>
        {data.map(row => (

              <td key={col.key}>
                {row[col.key]}
              </td>
            ))}
            <td>
              <QualityBadge score={row.quality_score} />
            </td>

              <ActionMenu row={row} />

          </tr>
        ))}
      </tbody>
    </table>
  );
}

export function QualityBadge({ score, grade }: { score: number; grade: 'A' | 'B' | 'C' }) {
  const colorMap = {
    A: '#4ade80',  // 绿色
    B: '#facc15',  // 黄色
    C: '#f87171'   // 红色
  };

  return (
    <span className="quality-badge" style={{ backgroundColor: colorMap[grade] }}>
      {grade}级 {score}/100
    </span>
  );
}
